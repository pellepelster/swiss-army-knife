{{- /* the link hook of relearn, with one more case: a relative link of a document of docs/development/ to a file outside of it (AGENT.md, the code) goes to the file on github, as it does there */}} {{- $path := .Destination }} {{- $htmlchars := dict """ `"` "'" `'` "+" `+` "&" `&` }} {{- range $k, $v := $htmlchars }} {{- $path = replace $path $k $v }} {{- end }} {{- $url := urls.Parse $path }} {{- if and (not $url.Scheme) (not (strings.HasPrefix $path "/")) (not (strings.HasPrefix $path "#")) .Page.File }} {{- $file := path.Join "docs/development" (path.Dir .Page.File.Path) $url.Path }} {{- if not (strings.HasPrefix $file "docs/development/") }} {{- $path = printf "%s/%s" site.Params.repositoryFiles $file }} {{- with $url.Fragment }}{{ $path = printf "%s#%s" $path . }}{{ end }} {{- end }} {{- end }} {{- partial "shortcodes/link.html" (dict "page" .Page "url" $path "title" .Title "content" .Text ) }}
